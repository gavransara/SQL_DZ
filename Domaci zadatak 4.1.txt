--2.Koliko puta je igrač u istoj sezoni igrao za dva različita kluba?
with cte as (
    select distinct season, player_id, team_id
    from player_match_stats pms join matches m on m.match_id = pms.match_id
    order by player_id
)
select count(*) as no_diff_teams_same_player
from cte c1 join cte c2 on c1.season = c2.season and c1.player_id = c2.player_id
and c1.team_id < c2.team_id --nije <> da se ne bi duplirali
 --35

--3.Koliko utakmica ima više kartona od prosečne utakmice? Bez kartona računajte sa 0.
with cards_per_game as(
    select m.match_id, count(me.event_id) as no_cards
    from matches m left join match_events me on m.match_id = me.match_id 
    and event_type in ('yellow_card','second_yellow','red_card')
    group by m.match_id
)
select count(*) as cards_more_than_avg
from cards_per_game
where no_cards > (
                select 
                    avg(no_cards) as avg_cards_per_game
                from cards_per_game
                )
--370

--4.Koliko igrača je odigralo više od 10 utakmica u sezoni 2024/25?
with cte as (
    select pms.player_id, count(pms.match_id) as num_matches
    from player_match_stats pms join matches m on m.match_id = pms.match_id
    where m.season = '2024/25' and pms.minutes_played > 0 -- moze bez ovog uslova za minutes played ako se smatra da je igrao samo ako je bio u postavi za tu utakmicu
    group by pms.player_id
    having num_matches > 10
)
select count(player_id) as num_players
from cte 
--383
