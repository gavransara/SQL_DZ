--5.zadatak
select
    match_id,
    date(kickoff_ts),
    home_goals + away_goals as total_goals
FROM bootcamp.northern_star.matches_flat
where month(kickoff_ts) = 4 and year(kickoff_ts) = 2025
    and (home_goals + away_goals) between 2 and 4
order by date(kickoff_ts)    
--22

--6.zadatak
select
    match_id,
    date(kickoff_ts) as matchdate,
    home_team,
    away_team,
    concat(home_team_code, " ", home_goals_ht, ":", away_goals_ht, " ", away_team_code) as ht_result,
    concat(home_team_code, " ", home_goals, ":", away_goals, " ", away_team_code) as final_result
FROM bootcamp.northern_star.matches_flat
where home_goals_ht = away_goals_ht and home_goals <> away_goals
--244
