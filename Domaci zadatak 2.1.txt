--1.zadatak
select
    match_id,
    stadium_name,
    attendance,
    stadium_capacity,
    stadium_capacity - attendance as empty_seats
FROM bootcamp.northern_star.matches_flat
order by empty_seats desc
limit 5
--Zajednicko za prve tri je sto su na istom stadionu igrane

--2.zadatak
select
   distinct referee_name
FROM bootcamp.northern_star.matches_flat
where home_team = 'Riverton Athletic'
--15

--3.zadatak
select 
    match_id,
    home_goals_ht - away_goals_ht as ht_score,
    home_goals - away_goals as final_score
FROM bootcamp.northern_star.matches_flat
where away_goals_ht > home_goals_ht and away_goals < home_goals
--15

--4.zadatak
select
    match_id,
    home_team,
    away_team,
    home_goals_ht,
    away_goals_ht,
    home_goals,
    away_goals,
    away_goals - away_goals_ht + home_goals - home_goals_ht as goals_2nd_ht
FROM bootcamp.northern_star.matches_flat
order by goals_2nd_ht desc
limit 1

--da bismo dobili golove u drugom poluvremenu : 
--away_goals - away_goals_ht + home_goals - home_goals_ht
