-- 2. Kreiranje dimenzije datuma i činjenične tabele
-- Napravite tabelu dim_date na isti način kao na predavanju. 
CREATE OR REPLACE TABLE dim_date AS
SELECT CAST(DATE_FORMAT(d, 'yyyyMMdd') AS INT) AS date_key,
       d                                       AS full_date,
       DATE_FORMAT(d, 'EEEE')                  AS weekday_name,
       DAYOFWEEK(d) IN (1, 7)                  AS is_weekend,
       CASE WHEN MONTH(d) >= 7 THEN CONCAT(YEAR(d), '/', SUBSTRING(CAST(YEAR(d) + 1 AS STRING), 3, 2))
            ELSE CONCAT(YEAR(d) - 1, '/', SUBSTRING(CAST(YEAR(d) AS STRING), 3, 2))
       END                                     AS season
FROM (SELECT EXPLODE(SEQUENCE(DATE'2023-07-01', DATE'2025-06-30', INTERVAL 1 DAY)) AS d)

select * from dim_date

create or replace table fact_player_match as
select pms.match_id, date(kickoff_ts) as date_key, player_id, team_id, minutes_played, shots, passes, rating
from player_match_stats pms join matches m on pms.match_id = m.match_id
where status = 'played'

--3. Prosečna ocena igrača po poziciji 
--Koristeći napravljene dimenzije i činjeničnu tabelu, izračunajte prosečnu ocenu (rating) igrača za svaku poziciju tokom sezone 2024/25. Prikažite poziciju i odgovarajuću prosečnu ocenu.
with players_rating as(
    select dp.player_id, dp.position, rating
    from dim_player dp join fact_player_match fpm on dp.player_id = fpm.player_id join dim_date dd on dd.full_date = fpm.date_key
    where season = '2024/25' and rating <> 'nan'
)
select position, round(avg(rating), 2) as avg_rating
from players_rating
group by position

alter table fact_player_match add columns (total_goals int)

update fact_player_match fpm
set total_goals = (select count(event_id)
                    from match_events me
                    where fpm.player_id = me.player_id and fpm.match_id = me.match_id and event_type in ('goal', 'penalty_goal'))
