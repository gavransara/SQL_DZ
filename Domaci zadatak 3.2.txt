--Osnovni nivo
--1.Prikažite naziv svakog kluba i naziv stadiona na kojem igra domaće utakmice.
select t.team_name, s.stadium_name
from teams t left join stadiums s on t.stadium_id = s.stadium_id
--left join-om obuhvatamo i Harbor City FC koji nema svoj stadion, takodje, Brackley Road je domaci stadion za dva tima, Brackley Athletic i Brackley Harriers, a Dunmore Park za Dunmore Celtic i Dunmore County

--2.Prikažite ime i prezime igrača i naziv kluba za koji je igrao u sezoni 2024/25. Uklonite duplikate iz rezultata.
select distinct p.first_name, p.last_name, t.team_name
from matches m join player_match_stats pms on m.match_id = pms.match_id
join players p on pms.player_id = p.player_id
 join teams t on t.team_id = pms.team_id
where m.season = '2024/25'

--3.Prikažite datum odigravanja utakmice, naziv stadiona i broj gledalaca za sve utakmice u sezoni 2024/25.
select cast(m.kickoff_ts as date) as match_date , s.stadium_name, m.attendance
from matches m join stadiums s on m.stadium_id = s.stadium_id
where m.season = '2024/25'

--Srednji nivo
--1.Prikažite sve utakmice za koje nije evidentiran broj gledalaca. Prikažite datum, domaći klub i gostujući klub.
select cast(m.kickoff_ts as date) as match_date, t1.team_name as home_team, t2.team_name as away_team
from matches m join teams t1 on t1.team_id = m.home_team_id join teams t2 on t2.team_id = m.away_team_id
where m.attendance is null

--2.Pronađite stadione na kojima nije odigrana ni jedna utakmica u sezoni 2024/25.
select distinct s.stadium_name
from stadiums s left join matches m on m.stadium_id = s.stadium_id and m.season = '2024/25'
where m.stadium_id is null
--Fairhaven Lane

--3.Za svaki klub izračunajte ukupan broj žutih kartona koje su dobili njegovi igrači u sezoni 2024/25. Rezultat sortirajte opadajuće prema broju kartona.
select 
    t.team_name,
    sum(case when m.season = '2024/25' and me.event_type in ('yellow_card', 'second_yellow') then 1 else 0 end) as yellow_total
from teams t left join match_events me on t.team_id = me.team_id 
left join matches m on m.match_id = me.match_id  
group by t.team_name
order by yellow_total desc

--Napredni nivo
--1.Pronađite pet igrača sa najvećim brojem asistencija u sezoni 2024/25. Prikažite ime igrača i broj asistencija.
select  p.first_name,
        p.last_name,
        count(me.event_id) as no_assistance
from matches m join match_events me on m.match_id = me.match_id 
    join players p on me.related_player_id = p.player_id
where m.season = '2024/25' and me.related_player_id is not null
    and me.event_type = 'goal'
group by p.first_name, p.last_name
order by no_assistance desc
limit 5

--2.Pronađite tri igrača sa najvećim ukupnim brojem odigranih minuta u sezoni 2024/25. Prikažite ime igrača, klub i ukupan broj minuta.
select
    p.first_name,
    p.last_name,
    t.team_name,
    sum(minutes_played) as season_minutes_played
from player_match_stats pms join matches m on m.match_id = pms.match_id
join players p on pms.player_id = p.player_id
join teams t on pms.team_id = t.team_id
where m.season = '2024/25'
group by p.first_name, p.last_name, t.team_name
order by season_minutes_played desc
limit 13
-- prva cetiri igraca na listi imaju isti broj minuta - 3240; potom tri igraca imaju isti broj minuta - 3150; onda sest igraca ima isti broj minuta - 3060
--iz tog razloga je limit 13, kako bi svi bili obuhvaceni jer ne mozemo jednoznacno odrediti tacno tri igraca sa najvecom minutazom

--3.Za svaku sezonu prikažite broj isključenja po klubovima. Računajte događaje red_card i second_yellow. Uključite samo klubove koji su imali najmanje jedno isključenje.
select
    m.season,
    t.team_name,
    count(*) as no_exclusion
from matches m join match_events me on m.match_id = me.match_id
join teams t on t.team_id = me.team_id
where me.event_type in ('red_card', 'second_yellow')
group by m.season, t.team_name
having no_exclusion > 0 --mozemo i bez ovog uslova, zbog uslova u where koji daje samo iskljucenja
