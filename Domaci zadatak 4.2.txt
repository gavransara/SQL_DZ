--1.Сезона у бројевима: утакмице, победе, нерешене, порази, дати и примљени голови, бодови
WITH club AS (SELECT team_id FROM teams WHERE team_name = 'Castlebrook United')
SELECT 
    count(m.match_id) as num_matches,
    sum(case when (c.team_id = m.home_team_id and home_goals > away_goals) or 
    (c.team_id = m.away_team_id and home_goals < away_goals)
    then 1 else 0 end) as wins,
    sum(case when (c.team_id = m.home_team_id and home_goals < away_goals) or 
    (c.team_id = m.away_team_id and home_goals > away_goals)
    then 1 else 0 end) as losses,
    sum(case when (c.team_id = m.home_team_id and home_goals = away_goals) or 
    (c.team_id = m.away_team_id and home_goals = away_goals)
    then 1 else 0 end) as draw,
    sum(case when c.team_id = m.home_team_id then home_goals else away_goals end) as scored_goals,
    sum(case when c.team_id = m.home_team_id then away_goals else home_goals end) as 
    received_goals,
    sum(case when (c.team_id = m.home_team_id and home_goals > away_goals) or 
    (c.team_id = m.away_team_id and home_goals < away_goals) then 3
             when (c.team_id = m.home_team_id and home_goals = away_goals) or 
    (c.team_id = m.away_team_id and home_goals = away_goals) then 1
            else 0 end) as points
from club c left join matches m on c.team_id = m.home_team_id or c.team_id = m.away_team_id 
where season = '2024/25' and status = 'played'

--3.Стрелци: три најбоља стрелца клуба и најбољи асистент.
WITH club AS (SELECT team_id FROM teams WHERE team_name = 'Castlebrook United')
select 
    me.player_id,
    p.first_name,
    p.last_name,
    sum(case when me.event_type = 'penalty_goal' then 1 else 0 end) as num_penalty,
    sum(case when me.event_type = 'goal' then 1 else 0 end) as num_clear_goal,
    count(me.event_id) as no_goals
from club c join match_events me on c.team_id = me.team_id
    join matches m on (c.team_id = m.home_team_id or c.team_id = m.away_team_id) and me.match_id = m.match_id join players p on p.player_id = me.player_id
where m.season = '2024/25' and me.event_type in ('goal', 'penalty_goal') 
group by me.player_id, p.first_name, p.last_name
order by no_goals desc

--4.Дисциплина: картони клуба по утакмици у поређењу са просеком лиге по тиму по утакмици
--prosecan broj kartona tima, prosecan broj kartona svih timova
WITH club AS (SELECT team_id FROM teams WHERE team_name = 'Castlebrook United'),
cards_per_game as(
    select 
        m.match_id, 
        m.home_team_id as team_id,
        count(me.event_id) as no_cards
    from matches m left join match_events me on m.match_id = me.match_id 
    and event_type in ('yellow_card','second_yellow','red_card') and me.team_id = m.home_team_id
    where season = '2024/25' and m.status = 'played'
    group by m.match_id, m.home_team_id
    UNION ALL
    select 
        m.match_id, 
        m.away_team_id as team_id,
        count(me.event_id) as no_cards
    from matches m left join match_events me on m.match_id = me.match_id 
    and event_type in ('yellow_card','second_yellow','red_card') and me.team_id = m.away_team_id
    where season = '2024/25' and m.status = 'played'
    group by m.match_id, m.away_team_id
)
select 
    round(avg(case when team_id in (select team_id from club) then no_cards end), 2) as club_cards_per_match,
    round(avg(no_cards), 2) as league_cards_per_match
from cards_per_game

--Ja sam uzela da analiziram tim koji u tabeli manager_spells ima problem tj poklapanja menadzera u istom vremenskom periodu
--Kako bismo predupredili dupliranja zbog DQ greske, napravila sam view koji cu koristiti u ovom delu zadatka, kao sto smo pokazali na casu.

create or replace view manager_spells_clean as (
    select * from manager_spells
    where spell_id not in (27,28,29)
)

WITH club AS (SELECT team_id FROM teams WHERE team_name = 'Castlebrook United')
select 
    distinct ms.manager_name,
    count(m.match_id) as no_matches,
    round(sum(case when (c.team_id = m.home_team_id and m.home_goals > m.away_goals) or 
    (c.team_id = m.away_team_id and m.home_goals < m.away_goals) then 3
        when (c.team_id = m.home_team_id and m.home_goals = m.away_goals) or 
    (c.team_id = m.away_team_id and m.home_goals = m.away_goals) then 1
        else 0 end) / count(m.match_id), 2) as points_per_match
from club c join matches m on c.team_id = m.home_team_id or c.team_id = m.away_team_id join manager_spells_clean ms on ms.team_id = c.team_id
and date(kickoff_ts) between ms.start_date and coalesce(ms.end_date, '9999-12-31')
where m.season = '2024/25'
group by ms.manager_name

--6.Највећа победа и најтежи пораз
WITH club AS (SELECT team_id FROM teams WHERE team_name = 'Castlebrook United'),
stats as(
        select 
        c.team_id as our_team_id,
        case when c.team_id = m.home_team_id then away_team_id
            else home_team_id
            end as opponent_id,
        case when c.team_id = m.home_team_id then m.home_goals
            else m.away_goals
            end as our_team_goals,
        case when c.team_id = m.home_team_id then m.away_goals
            else m.home_goals
            end as opponent_goals,
         s.stadium_name   
    from club c join matches m on c.team_id = m.home_team_id or c.team_id = m.away_team_id join stadiums s on s.stadium_id = m.stadium_id
    where season = '2024/25' and status = 'played'
)
select 
    t1.team_name as our_team,
    t2.team_name as opponent,
    s.our_team_goals,
    s.opponent_goals,
    s.our_team_goals - s.opponent_goals as difference,
    s.stadium_name
from stats s join teams t1 on s.our_team_id = t1.team_id join teams t2 on s.opponent_id = t2.team_id
order by difference desc

--7.Публика: просечна посета и попуњеност на домаћим утакмицама, и колико утакмица нема податак о посети.
WITH club AS (SELECT team_id FROM teams WHERE team_name = 'Castlebrook United')
select  
        round(avg(m.attendance), 0) as avg_attendance,
        round(avg(m.attendance * 100 / s.capacity), 2) as occupancy,
        sum(case when attendance is null then 1 else 0 end) as unknown_attendance
from club c join matches m on c.team_id = m.home_team_id join stadiums s on m.stadium_id = s.stadium_id
where season = '2024/25' and status = 'played'
